## day02\_02\_status.py

```shell
from flask import Flask

app = Flask(__name__)

@app.route('/')
def index():

    return 'index'

###############设置cookie
"""

概念 -- 项目中的使用 -- 自己遇到的问题或者是自己的个人见解

第一次:  用户通过浏览器第一次访问我们的网址.这个时候 请求不会携带任何cookie信息
        当我们的服务器接收到请求之后,会设置cookie. cookie就以响应头的形式 返回给客户端.客户端保存cookie
        
第二次及其之后的请求: 
         用户通过浏览器请求,都会携带cookie信息
         我们的服务器就可以获取cookie数据

"""
from flask import make_response,request

@app.route('/set_cookie')
def set_cookie():

    #模拟获取用户信息
    # set_cookie?username=itcast
    name=request.args.get('username','')

    response = make_response('set_cookie')

    #方法是和django一样的
    response.set_cookie(key='name',value=name,max_age=3600)

    # response.delete_cookie(key='name')

    return response

@app.route('/get_cookie')
def get_cookie():

    name = request.cookies.get('name','')

    return 'get_cookie  {}'.format(name)

############################################session

"""
第一次:  用户通过浏览器第一次访问我们的网址.这个时候 请求不会携带任何cookie信息 特别是 sessionid
        当我们的服务器接收到请求之后,会设置session. 设置成功之后,数据会保存.然后会生成一个 sessionid 的cookie信息
         cookie就以响应头的形式 返回给客户端.客户端保存cookie
        
第二次及其之后的请求: 
         用户通过浏览器请求,都会携带cookie信息 特别是 sessionid
         我们的服务器就可以获取cookie中的 sessionid了  根据sessionid来判断读取数据

"""

# SECRET_KEY 用于session数据加密
app.config['SECRET_KEY']='abc'

from flask import session,request

@app.route('/set_session')
def set_session():

    # 模拟用户登录
    username=request.args.get('username','')
    # session[key]=value

    session['username']=username

    return 'set_session'

@app.route('/get_sesson')
def get_session():

    username = session['username']

    return 'get session {}'.format(username)

if __name__ == '__main__':

    app.run()
```

```shell
from flask import Flask,request
import re
app=Flask(__name__)

@app.route('/')
def index():

    username = request.args.get('username','')

    # django方式 可以的
    # if not re.match():
    #     return {},400

    from flask import abort
    # flask 的方式
    if not re.match('[a-zA-Z0-9_-]{5,20}',username):
        # abort 字面意思就是 中断 不在执行后续代码
        # abort(响应状态码)
        # 响应状态码 必须是 http规定的
        abort(400)

    return 'index'

# 针对于  abort 的响应状态码 我们可以做统一的处理
# 例如 只要是 响应状态码为 400 都走这里

@app.errorhandler(400)
def params_error(e):
    # 函数 必须要有 形参 这个形参是获取的 error 错误信息
    return '参数错误'

if __name__ == '__main__':
    app.run()
```

## day02\_07\_request.py

```shell
from flask import Flask
from flask_restful import Resource,Api

app = Flask(__name__)
api=Api(app)

#################定义验证手机号的函数
import re
def check_mobile(value):

    if not re.match('1[3-9]\d{9}',value):
        raise Exception('数据不满足要求')

    return value

class RegisterResouce(Resource):

    def post(self):

        """
        1. 导入库
        2. 创建实例对象
        3. 添加数据验证 和验证的参数
        4. 调用验证方法

        :return:
        """
        # 1. 导入库
        from flask_restful import reqparse
        # 2. 创建实例对象
        rp = reqparse.RequestParser()
        # 3. 添加数据验证 和验证的参数
        # add_argument 添加要验证的字段

        # ① 字段的名字是什么呢??? key  第一个参数: key

        # ② location 位置 . 你要验证的 字段的数据在哪里???  args,form,json,headers,cookies

        # ③ required = True 就是必须要传递的

        # ④ help  设置提示信息

        # ⑤ type 函数名 对 指定的数据 进行额外的方法验证
        rp.add_argument('username',location='json',required=True)
        rp.add_argument('mobile',location='json',required=True,type=check_mobile)
        rp.add_argument('password',location='json',required=True,help='password is required')

        # 4. 调用验证方法
        # parse_args() 会把 验证满足需求的数据 再给我们
        data = rp.parse_args()

        return data

        return {'msg':'register'}

api.add_resource(RegisterResouce,'/register')

@app.route('/')
def index():
    return 'index'

if __name__ == '__main__':
    app.run()
```

## day02\_08\_response.py

```shell
from flask import Flask
from flask_restful import Resource,Api

app = Flask(__name__)
api=Api(app)

#模拟对象
class Student:

    def __init__(self,id,name,age):
        self.id=id
        self.name=name
        self.age=age

# 1. 导入
from flask_restful import fields,marshal_with,marshal
# 2. 定义字段
stu_fields = {
    'id':fields.Integer,
    'name':fields.String
}

class LoginResource(Resource):

    @marshal_with(stu_fields,envelope='data')
    def get(self):
        # 3. 返回响应
        return Student(id=666,name='itcast',age=15)
        # return {'msg':'login'}

    def post(self):
        stu = Student(id=666,name='itcast',age=15)
        return marshal(stu,stu_fields,envelope='info')

api.add_resource(LoginResource,'/login')

@app.route('/')
def index():
    # string,dict,tulpe
    return 'index'

from flask_restful.representations import json

from flask import make_response, current_app
from flask_restful.utils import PY3
from json import dumps

# 如何让系统调用我们这个方法
# Content-Type  MIME

@api.representation('application/json')
def output_json(data, code, headers=None):
    """Makes a Flask response with a JSON encoded body"""

    settings = current_app.config.get('RESTFUL_JSON', {})

    # If we're in debug mode, and the indent is not set, we set it to a
    # reasonable value here.  Note that this won't override any existing value
    # that was set.  We also set the "sort_keys" value.
    if current_app.debug:
        settings.setdefault('indent', 4)
        settings.setdefault('sort_keys', not PY3)

    # always end the json dumps with a new line
    # see https://github.com/mitsuhiko/flask/pull/1262

    data = {"message": "ok", "eeeeeeeee": data}

    dumped = dumps(data, **settings) + "\n"

    resp = make_response(dumped, code)
    resp.headers.extend(headers or {})
    return resp

if __name__ == '__main__':
    app.run()
```

```shell
# 定义类视图
class AdminLoginResouce(Resource):

    def get(self):
        return {'msg':'get'}

def login_required(func):

    def wrapper(*args,**kwargs):

        #伪代码
        isLogin=False

        if isLogin:
            return func(*args,**kwargs)
        else:
            return {'msg':'please login'}

    return wrapper

# 定义一个个人中心类视图
class CenterResource(Resource):

    # 所有的方法 都执行装饰器
    # method_decorators = [
    #     login_required,
    # ]

    # 就是 post需要登录
    method_decorators = {
        'post': [login_required,]
    }

    def get(self):

        return {'msg':'get center'}

    def post(self):

        return {'msg': 'post center'}

# 添加 类视图的 路由
admin_api.add_resource(AdminLoginResouce,'/admin/login')
admin_api.add_resource(CenterResource,'/admin/center')
```
