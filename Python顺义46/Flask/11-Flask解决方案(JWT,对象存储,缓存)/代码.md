# https://developer.qiniu.com/kodo/1242/python

# -*- coding: utf-8 -*-
# flake8: noqa

from qiniu import Auth, put_file

#需要填写你的 Access Key 和 Secret Key
#AK:q9crPZPROOXrykaH85q_zpEEll0f_LsjXwUnXHRo
# SK:lG_4_tI8bJTR8Zk6z8fGwYp79aQHkJgolvvBL_qm

access_key = 'q9crPZPROOXrykaH85q_zpEEll0f_LsjXwUnXHRo'
secret_key = 'lG_4_tI8bJTR8Zk6z8fGwYp79aQHkJgolvvBL_qm'

#1. 构建鉴权对象   -- 七牛云要验证我们的 ak和sk
q = Auth(access_key, secret_key)

#2. 要上传的空间   -- 刚才创建的空间
bucket_name = 'shanghai41'

#3.上传后保存的文件名  -- 暂时先不管
key = 'dddd.png'

#4.生成上传 Token，可以指定过期时间等   -- 其实就是为了验证
token = q.upload_token(bucket_name, key, 3600)

#5.要上传文件的本地路径   -- 其实它是要 本地图片上传  给一个绝对路径 或者 相对路径
localfile = '/home/ubuntu/Desktop/img/dddd.png'

#6.把本地图片上传到七牛云
# 上传之后,会返回信息
ret, info = put_file(token, key, localfile, version='v2')
# print(ret)
# {'hash': 'FtFixpV8UBwZqgv2JgviRuR9ykuw', 'key': 'my-python-logo.png'}
# print(info)
from qiniu.http import ResponseInfo
import json
if info.ok() :
    #说明上传成功
    print('上传成功')
    # info.text_body   是 str
    key=json.loads(info.text_body).get('key')
    print(key)
