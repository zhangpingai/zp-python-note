## 04-script.py

```shell
from flask import Flask

# 1. 导入 Manager
from flask_script import Manager

app=Flask(__name__)

# 2. 创建manager实例对象
# 传递flask实例对象
manager=Manager(app=app)
@app.route('/')
def index():
    return 'script'

# flask-script 要配合传统的运行方式来
if __name__ == '__main__':
    # 3. 让manager运行起来
    # app.run()
    manager.run()
```

## 07-request.py

```shell
from flask import Flask

app=Flask(__name__)

@app.route('/')
def index():

    return 'index'

# 路径中的参数
# goods/id
# goods/123

@app.route('/goods/<int:id>')
def goods_detail(id):

    return 'detail {}'.format(id)

# 1. 导入 BaseConverter
from werkzeug.routing import BaseConverter

# BaseConverter
# 2. 继承并重写 regex

class MobileConverter(BaseConverter):

    regex = '1[3-9]\d{9}'

# 3. 告知系统我们的自定义转换器
# app.url_map.converters[key]=className
app.url_map.converters['phone']=MobileConverter

# register/mobile
@app.route('/register/<phone:mobile>')
def register(mobile):

    return 'phone is {}'.format(mobile)

########获取查询字符串#########################
from flask import request

@app.route('/search')
def search():

    print(request.args)
    # ImmutableMultiDict([('keyword', 'abc')])
    return 'search'

############form 表单####################

@app.route('/login',methods=['POST'])
def login():

    print(request.form)
    # ImmutableMultiDict([('username', 'itcast'), ('password', '123')])

    return 'login'

#############JSON数据#########################
@app.route('/json',methods=['POST'])
def json():

    print(request.json)
    # {'username': 'itcast', 'password': '123'}

    return 'json'

if __name__ == '__main__':
    app.run()
```
