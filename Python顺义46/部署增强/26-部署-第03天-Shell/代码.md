## case.sh

```shell
#!/bin/bash

<<mark
case var in 
	pattern1)
		如果var 满足pattern1 则执行这里 执行了之后就不再执行其他的
		如果var 不满足，则继续匹配
	;;

	pattern2)
		xxxx
	;;

	pattern3)
		xxxxx
	;;

	*)
		xxxxx
	;;
esac

mark

# 把刚才的 if 改成 case

cmd="$1"

case $cmd in
	"stop")
		echo "stopping...."
	;;

	"start")
		echo "starting...."
	;;

	"restart")
		echo "restarting...."
	;;

	*)
		echo "please input start|stop|restart"
	;;
esac
```

## for.sh

```shell
#!/bin/bash

<<mark
for i in {1..10}
do
	echo 'good'
done
mark

<<mark2
# ls /home 罗列home目录下的文件
# 遍历 指定目录下的文件
for item in $(ls /home)
do
	echo $item
done
mark2

# 累加 10以内的数值

sum=0
for i in {1..10}
do
	if [ $i -eq 5 ]
	then
		#break
		exit
	fi
	let sum=sum+i
done

echo $sum
```

## service.sh

#!/bin/bash

cmd="$1"
# /bin/bash service.sh start|stop|restart
if [ $cmd == 'start' ]
then
	echo "service is starting...."
elif [ $cmd = 'stop' ]
then
	echo "service is stopping...."
elif [ $cmd == 'restart' ] 
then
	echo "service is restarting...."
else
	echo "please input correct command start|stop|restart"
fi
