## meiduo.conf

```shell
upstream meiduo {
	server 192.168.19.130:8001;
	server 192.168.19.130:8002;
}

server {
	server_name www.meiduo.site;
	listen 80;
	
	location =/ {
		root /data/meiduo/front_page/;
		index index.html;
		try_files $uri $uri/ =404;
	}

	location /static {
		alias /data/meiduo/front_page/;
	}

	location ~ \.html$ {
		root /data/meiduo/front_page/;
	}

	location / {
		include uwsgi_params;
		uwsgi_pass meiduo;
	}
}

```
